# SMS port key bindings for controller 1.
# Format: CONTROL = KEY [KEY ...]   (a line replaces that control's defaults)
# Controls: A B X Y Z L R START DPAD_UP DPAD_DOWN DPAD_LEFT DPAD_RIGHT
#           STICK_UP STICK_DOWN STICK_LEFT STICK_RIGHT
#           CSTICK_UP CSTICK_DOWN CSTICK_LEFT CSTICK_RIGHT HALF_TILT QUIT
#           SAVE_STATE LOAD_STATE (save a save state, load the newest)
#           MAP (open or close the radar's full map)
#           L_SOFT R_SOFT (a light L or R press, no click; SMS_SOFT_TRIGGER
#           sets how far, default 95%)
# Keys: letters, digits, ENTER ESC SPACE TAB BACKSPACE UP DOWN LEFT RIGHT
#       LSHIFT RSHIFT LCTRL RCTRL LALT RALT KP_0..KP_9 KP_ENTER F1..F12
#       COMMA PERIOD SLASH SEMICOLON ... or #<SDL scancode>
# Mouse buttons count as keys: MOUSE_LEFT MOUSE_RIGHT MOUSE_MIDDLE
#       MOUSE_BACK MOUSE_FORWARD (the side buttons), e.g. "A = MOUSE_LEFT X"
# Controller buttons (named for an Xbox pad; SDL maps others to the same
# places): PAD_A PAD_B PAD_X PAD_Y PAD_LB PAD_RB PAD_LT PAD_RT PAD_START
#       PAD_BACK PAD_GUIDE PAD_LSTICK PAD_RSTICK PAD_DPUP PAD_DPDOWN
#       PAD_DPLEFT PAD_DPRIGHT PAD_MISC PAD_PADDLE1..4 PAD_TOUCHPAD
# A line may list keys, controller buttons or both: its keys replace the
# control's keys and its buttons replace its controller buttons, so
# "A = PAD_Y" keeps the left click. The controller defaults: A B X Y = PAD_A
# PAD_B PAD_X PAD_Y, Z = PAD_RB, L R = PAD_LT PAD_RT (analog), START =
# PAD_START, DPAD_* = PAD_DP*, SAVE_STATE = PAD_LB+PAD_RB, MAP = PAD_BACK.
# The sticks always move the sticks. FIRST+SECOND (PAD_LB+PAD_RB, LCTRL+S) is
# two keys or two buttons held together.
# Loaded from ./bindings.txt (run.sh starts the game in this folder; or
# ../../bindings.txt when started from build/<os>-<arch>/),
# or from the path in SMS_BINDINGS. These are the built-in defaults, for
# keyboard and mouse with mouse look (mouse_camera): WASD moves, the mouse
# turns the camera, left click jumps, right click dives, Q switches the nozzle,
# E is the close-up camera, middle click sprays FLUDD softly and Space fully,
# Left Shift is L (ground pound, sidestep) and Tab the map (Z).
A = MOUSE_LEFT
B = MOUSE_RIGHT
X = Q
Y = E
Z = TAB
L = LSHIFT
R = SPACE
R_SOFT = MOUSE_MIDDLE
START = ENTER
DPAD_UP = 1 KP_8
DPAD_DOWN = 2 KP_2
DPAD_LEFT = 3 KP_4
DPAD_RIGHT = 4 KP_6
STICK_UP = W UP
STICK_DOWN = S DOWN
STICK_LEFT = A LEFT
STICK_RIGHT = D RIGHT
CSTICK_UP = I
CSTICK_DOWN = K
CSTICK_LEFT = J
CSTICK_RIGHT = L
HALF_TILT = LCTRL
QUIT = ESCAPE
SAVE_STATE = F5
LOAD_STATE = F6
MAP = M
